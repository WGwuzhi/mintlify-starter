## 创建 worker

保存为 `plugin/adapters/adapter_demo_http.js`。下面使用 Node.js 22 内置 `WebSocket`，是生命周期骨架；实现平台处理后再声明对应能力。

```js
//[title: adapter_demo_http]
//[author: your-name]
//[version: 1.0.0]
//[class: 适配器]
//[language: javascript]
//[runtime: node@22.22.3]
//[description: Demo IM 适配器]
//[im_type: demo]
//[im_type_name: Demo IM]
//[protocol: im]
//[transport: http]
//[account_id_key: bot_id]
//[param: {"scope":"account","required":true,"key":"bot_id","name":"Bot ID"}]

'use strict';

const IM_TYPE = 'demo';
const CONTROL_URL = process.env.AUTCLAW_ADAPTER_CONNECT_URL
  || `ws://127.0.0.1:${process.env.AUTCLAW_PORT || '8200'}/api/adapters/${IM_TYPE}/connect`;
let generation = 0;

function object(value) {
  return value !== null && typeof value === 'object' && !Array.isArray(value);
}
function send(session, packet) {
  if (session.closed || session.ws.readyState !== WebSocket.OPEN) {
    throw new Error('control connection closed');
  }
  session.ws.send(JSON.stringify(packet));
}
function call(session, action, params, timeoutMS = 15000) {
  if (session.pending.size >= 128) return Promise.reject(new Error('worker busy'));
  return new Promise((resolve, reject) => {
    const echo = `node_${session.generation}_${++session.seq}`;
    const timer = setTimeout(() => {
      session.pending.delete(echo);
      reject(new Error(`${action} timed out`));
    }, timeoutMS);
    session.pending.set(echo, { resolve, reject, timer });
    try {
      send(session, { aut_action: action, aut_echo: echo, aut_params: params });
    } catch (error) {
      clearTimeout(timer);
      session.pending.delete(echo);
      reject(error);
    }
  });
}
function reply(session, echo, params, error) {
  send(session, error
    ? { aut_echo: echo, ok: false, aut_error: String(error.message || error) }
    : { aut_echo: echo, ok: true, aut_params: params || {} });
}
function detach(session) {
  if (session.closed) return;
  session.closed = true;
  session.abort.abort();
  for (const entry of session.pending.values()) {
    clearTimeout(entry.timer);
    entry.reject(new Error('control connection closed'));
  }
  session.pending.clear();
  session.ws.close();
  setTimeout(connect, 3000);
}
function connect() {
  const session = {
    generation: ++generation,
    ws: new WebSocket(CONTROL_URL),
    pending: new Map(), accounts: new Map(),
    abort: new AbortController(),
    closed: false, initialized: false, seq: 0, active: 0,
  };
  session.ws.addEventListener('message', ({ data }) => {
    let packet;
    try {
      packet = JSON.parse(String(data));
      if (!object(packet)) throw new Error('invalid packet');
    } catch {
      detach(session);
      return;
    }
    onPacket(session, packet).catch(() => detach(session));
  });
  session.ws.addEventListener('close', () => detach(session));
  session.ws.addEventListener('error', () => detach(session));
}
async function onPacket(session, packet) {
  if (session.closed) return;
  const echo = typeof packet.aut_echo === 'string' ? packet.aut_echo : '';
  if (!packet.aut_action && echo) {
    const entry = session.pending.get(echo);
    if (!entry) return;
    session.pending.delete(echo);
    clearTimeout(entry.timer);
    if (packet.ok === false) entry.reject(new Error(packet.aut_error || 'call failed'));
    else if (!object(packet.aut_params)) entry.reject(new Error('invalid response'));
    else entry.resolve(packet.aut_params);
    return;
  }
  if (!object(packet.aut_params)) throw new Error('invalid aut_params');
  const params = packet.aut_params;
  if (packet.aut_action === 'adapter_init') {
    if (session.initialized) throw new Error('duplicate init');
    session.accounts = new Map((params.account_configs || []).map(config => {
      if (!config.account_id || !config.instance_id) throw new Error('invalid account');
      return [String(config.account_id), config];
    }));
    session.hostCapabilities = params.host_capabilities || {};
    session.globalConfig = params.global_config_values || {};
    send(session, {
      aut_action: 'adapter_worker_ready',
      aut_params: {
        contract: 'v6',
        resolve_account: false, normalize_inbound: false,
        build_outbound: false, decode_receipt: false,
        execute_outbound: false, emit_inbound_events: false,
        request_outbound: false, open_ingress: false, supports_qr_login: false,
        features: [], outbound_actions: [],
      },
    });
    session.initialized = true;
    // 在这里启动平台循环，使用 session.abort.signal 取消平台请求。
    return;
  }
  if (!echo) return;
  if (!session.initialized || session.active >= 8) {
    reply(session, echo, null, new Error('worker not ready or busy'));
    return;
  }
  session.active++;
  try {
    const result = await handleAction(session, packet.aut_action, params);
    if (!object(result)) throw new Error('invalid handler result');
    reply(session, echo, result);
  } catch (error) {
    if (!session.closed) reply(session, echo, null, error);
  } finally {
    session.active--;
  }
}
async function handleAction(session, action, params) {
  throw new Error(`unsupported control action: ${action}`);
}
connect();
```

每个连接持有独立的账号快照、pending 和取消信号。先处理响应再分派请求，保证 handler 内 `await call(...)` 时仍能接收响应。平台请求同时设置超时，例如 `AbortSignal.any([session.abort.signal, AbortSignal.timeout(10000)])`。

## 实现平台动作

主动上报、自执行发送的 worker 完成四项核心 codec 和 execute 后，将对应 ready 布尔值设为 `true`，并声明：

```js
inbound_events: ['message'],
features: ['message.proactive'],
outbound_actions: ['message.send'],
```

在 `handleAction` 中分派：

- `adapter_codec_resolve_account`：检查原始请求与账号索引，返回 `{ account_id }`。
- `adapter_codec_normalize_inbound`：返回 `{ events }`，使用请求的 `instance_id`。
- `adapter_codec_build_outbound`：检查 `params.action` 与目标，转换 `params.payload`；自发送返回 `{ executor: 'worker', transport: 'ws', await_receipt: false, raw_payload }`。
- `adapter_codec_decode_receipt`：返回 `{ matched: false }`，或回执的 `matched`、`request_id`、`ok`、`data`。
- `adapter_execute_outbound`：按 `params.account_id` 取凭据，发送 `params.raw_payload`。平台失败返回 `{ ok: false, error }`，成功返回 `{ ok: true, data }`。

按 [动作映射](/cn/plugin-sdk/adapter#构造出站与能力映射)实现 `sendText`、`stopTyping` 等 worker 动作，能力名和平台 API 名分别处理。

## 上报事件与游标

按 [标准事件](/cn/plugin-sdk/adapter#上报标准事件)构造 `events`，填入当前账号配置的 `instance_id`：

```js
const result = await call(session, 'adapter_emit_inbound_events', { im_type: IM_TYPE, events });
const advance = advanceCount(result, events.length);
// 只保存 events.slice(0, advance) 对应的平台游标。

function advanceCount(result, total) {
  const outcomes = new Map((result.outcomes || []).map(item => [item.index, item]));
  const accepted = new Set(['queued', 'waiter_consumed', 'filtered', 'duplicate']);
  for (let index = 0; index < total; index++) {
    if (!accepted.has(outcomes.get(index)?.status)) return index;
  }
  return total;
}
```

超时或断线时不推进游标，重试未确认批次；拒绝时停在缺口前，重试缺口与后续事件。保持事件 ID 和 `dedupe_key` 稳定。平台 ID 使用字符串，避免超过 JavaScript 安全整数范围。

## 媒体与回执

媒体使用 `MediaRef`。需要上传文件时，先检查 `session.hostCapabilities.media_resolve`，在当前 build 内调用：

```js
const resolved = await call(session, 'adapter_host_media_resolve', {
  execution_id: params.execution_id,
  segment_type: 'file',
  media: params.payload.media,
  representation: 'file_path',
  target: { format: 'original' },
});
// 在本次执行结束前读取 resolved.file_path 并上传。
```

execute 分支从自己构造的 `raw_payload` 读取媒体，保留解析需要的 `MediaRef`。复用当前 `execution_id`，不跨执行缓存临时资源。

支持撤回的发送返回：

```js
return {
  ok: true,
  data: {
    sent_messages: messageIDs.map(message_id => ({
      message_id,
      recall_context: { chat_id: chatID },
    })),
  },
};
```

部分发送失败仍保留成功项。处理 `recallMessage` 时读取 `payload.message_id` 和 `payload.recall_context`。
